# requirements-media.txt — video / audio / YouTube → quiz (services/media_quiz/)
# Optional add-on to requirements.txt; without it the media endpoints return 503
# and the document quiz keeps working.
#
#   pip install -r requirements.txt -r requirements-media.txt
#
# No PyTorch and no ffmpeg binary needed: PyAV decodes media, Silero VAD ships
# inside faster-whisper as ONNX, RapidOCR runs PaddleOCR's models on ONNX Runtime.
# First run downloads the Whisper model (MEDIA_WHISPER_MODEL, default "small",
# ~480 MB) and the RapidOCR models (bundled in the wheel).
# ─────────────────────────────────────────────────────────────────────────────

faster-whisper>=1.1.0          # ASR (CTranslate2) + bundled Silero VAD + PyAV decoding
av>=12.0.0                     # container / stream probing (pulled in by faster-whisper)
opencv-python-headless>=4.9    # frame sampling, scene-change detection
rapidocr_onnxruntime>=1.3.0    # text detection (probe) + recognition (keyframes)
# NOTE: rapidocr depends on the FULL opencv-python wheel, which wins over the
# headless one above and needs libGL / libglib at import time. On a headless
# server: sudo apt-get install -y libgl1 libglib2.0-0 (deploy/oracle/update.sh
# does this automatically), or video uploads 503 while audio still works.
# YouTube links. Keep this recent: YouTube's anti-bot checks change often and
# only new releases can get past them.
yt-dlp>=2026.8.19
